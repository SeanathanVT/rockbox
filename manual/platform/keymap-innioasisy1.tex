% Button-name and per-context action macros for the Innioasis Y1.
% Derived from apps/keymaps/keymap-innioasisy1.c. The Y1 is a click-wheel
% device (top/centre/bottom + left/right + a capacitive scroll ring); its top
% button is "Back" (short) / context menu (long), the centre is "Select", the
% bottom is "Play". It has no Hold switch, FM radio, recording, or RTC alarm,
% so those action macros are placeholders for build-safety and are not used
% (the corresponding manual sections are not enabled in innioasisy1.tex).

\newcommand{\ButtonLeft}{\btnfnt{Left}}
\newcommand{\ButtonRight}{\btnfnt{Right}}
\newcommand{\ButtonMenu}{\btnfnt{Back}}
\newcommand{\ButtonPlay}{\btnfnt{Play}}
% No dedicated power button; power-off is a long press of Play.
\newcommand{\ButtonPower}{\ButtonPlay}
\newcommand{\ButtonSelect}{\btnfnt{Select}}
\newcommand{\ButtonScrollFwd}{\btnfnt{Scroll Forward}}
\newcommand{\ButtonScrollBack}{\btnfnt{Scroll Backward}}
% No physical up/down or Hold; 2D plugins use the scroll ring for up/down.
\newcommand{\ButtonUp}{\ButtonScrollBack}
\newcommand{\ButtonDown}{\ButtonScrollFwd}
\newcommand{\ButtonHold}{\btnfnt{Hold}}
